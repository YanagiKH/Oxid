
use std::io::write_stdout as output;
use std::io::WriteStatus as Status;
fn main() -> i32 {
    let bytes = [65, 66, 256];
    let status = output(&bytes);
    match status {
        Status::Complete => { return 0; },
        Status::InvalidInput => { return 1; },
        Status::IoError(n) => { return n + 2; },
    }
}
